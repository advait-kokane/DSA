struct Node {
    int prn;
    char name[50];
    struct Node *next;
};

struct Node* create(int prn, char name[]) {
    struct Node *p = malloc(sizeof(struct Node));
    p->prn = prn;
    strcpy(p->name, name);
    p->next = NULL;
    return p;
}

void addPresident(struct Node **head, int prn, char name[]) {
    struct Node *p = create(prn, name);
    p->next = *head;
    *head = p;
}

void addSecretary(struct Node **head, int prn, char name[]) {
    struct Node *p = create(prn, name);

    if (*head == NULL) {
        *head = p;
        return;
    }

    struct Node *t = *head;
    while (t->next != NULL)
        t = t->next;

    t->next = p;
}

void addMember(struct Node *head, int prn, char name[]) {
    if (head == NULL) {
        printf("Add President first\n");
        return;
    }

    struct Node *p = create(prn, name);
    p->next = head->next;
    head->next = p;
}

void deletePresident(struct Node **head) {
    if (*head == NULL) {
        printf("List is empty\n");
        return;
    }

    struct Node *t = *head;
    *head = (*head)->next;
    free(t);
}

void deleteSecretary(struct Node **head) {
    if (*head == NULL) {
        printf("List is empty\n");
        return;
    }

    if ((*head)->next == NULL) {
        free(*head);
        *head = NULL;
        return;
    }

    struct Node *t = *head;
    while (t->next->next != NULL)
        t = t->next;

    free(t->next);
    t->next = NULL;
}

void deleteMember(struct Node **head, int prn) {
    struct Node *t = *head;

    while (t != NULL && t->next != NULL && t->next->prn != prn)
        t = t->next;

    if (t == NULL || t->next == NULL) {
        printf("Member not found\n");
        return;
    }

    struct Node *d = t->next;
    t->next = d->next;
    free(d);
}

int countMembers(struct Node *head) {
    int count = 0;
    while (head != NULL) {
        count++;
        head = head->next;
    }
    return count;
}

void display(struct Node *head) {
    while (head != NULL) {
        printf("%d %s\n", head->prn, head->name);
        head = head->next;
    }
}

struct Node* concatenate(struct Node *a, struct Node *b) {
    if (a == NULL) return b;

    struct Node *t = a;
    while (t->next != NULL)
        t = t->next;

    t->next = b;
    return a;
}

int isOperator(char c) {

           c == '/' || c == '^';

int priority(char c) {
    if (c == '^') return 3;
    if (c == '*' || c == '/') return 2;
    return 1;
}

int power(int a, int b) {
    int result = 1;
    while (b > 0) {
        result = result * a;
        b--;
    }
    return result;
}

#include <stdio.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node* insert(struct Node *root, int x) {
    if (root == NULL) {
        struct Node *p = malloc(sizeof(struct Node));
        p->data = x;
        p->left = p->right = NULL;
        return p;
    }

    if (x < root->data)
        root->left = insert(root->left, x);
    else
        root->right = insert(root->right, x);

    return root;
}

int height(struct Node *root) {
    if (root == NULL) return 0;

    int left = height(root->left);
    int right = height(root->right);

    return (left > right ? left : right) + 1;
}

int minimum(struct Node *root) {
    while (root->left != NULL)
        root = root->left;

    return root->data;
}

int search(struct Node *root, int x) {
    if (root == NULL) return 0;
    if (root->data == x) return 1;

    return search(root->left, x) || search(root->right, x);
}

void mirror(struct Node *root) {
    if (root == NULL) return;

    struct Node *temp = root->left;
    root->left = root->right;
    root->right = temp;

    mirror(root->left);
    mirror(root->right);
}

void inorder(struct Node *root) {
    if (root == NULL) return;

    inorder(root->left);
    printf("%d ", root->data);
    inorder(root->right);
}

void dfs(int v) {

    visited[v] = 1;

    for (i = 0; i < n; i++) {
        if (graph[v][i] != 0 && !visited[i])
            dfs(i);
    }
}
